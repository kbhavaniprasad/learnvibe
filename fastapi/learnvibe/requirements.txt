from sqlalchemy import create_engine, Column, Integer, String, Float, DateTime, Text, JSON, Boolean, ForeignKey
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker, relationship
from sqlalchemy.dialects.postgresql import UUID
import uuid
from datetime import datetime

from config import settings

# Database connection
engine = create_engine(settings.database_url)
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
Base = declarative_base()

# Database Models

class User(Base):
    __tablename__ = "users"
    
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    email = Column(String, unique=True, index=True)
    username = Column(String, unique=True, index=True)
    hashed_password = Column(String)
    full_name = Column(String)
    is_active = Column(Boolean, default=True)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    
    # Relationships
    assessments = relationship("Assessment", back_populates="user")
    progress_records = relationship("ProgressRecord", back_populates="user")

class Goal(Base):
    __tablename__ = "goals"
    
    id = Column(String, primary_key=True)  # e.g., "data_scientist"
    name = Column(String, nullable=False)  # e.g., "Data Scientist"
    description = Column(Text)
    is_active = Column(Boolean, default=True)
    created_at = Column(DateTime, default=datetime.utcnow)
    
    # Relationships
    levels = relationship("CustomLevel", back_populates="goal")
    prerequisites = relationship("Prerequisite", back_populates="goal")

class CustomLevel(Base):
    __tablename__ = "custom_levels"
    
    id = Column(String, primary_key=True)
    goal_id = Column(String, ForeignKey("goals.id"))
    name = Column(String, nullable=False)
    description = Column(Text)
    experience_years = Column(String)
    required_skills = Column(JSON)  # List of skill IDs
    hierarchy_order = Column(Integer)  # For ordering levels
    created_at = Column(DateTime, default=datetime.utcnow)
    
    # Relationships
    goal = relationship("Goal", back_populates="levels")

class Prerequisite(Base):
    __tablename__ = "prerequisites"
    
    id = Column(String, primary_key=True)
    goal_id = Column(String, ForeignKey("goals.id"))
    name = Column(String, nullable=False)
    description = Column(Text)
    importance = Column(String)  # Critical, Important, Advanced
    related_topics = Column(JSON)  # List of related topics
    created_at = Column(DateTime, default=datetime.utcnow)
    
    # Relationships
    goal = relationship("Goal", back_populates="prerequisites")
    questions = relationship("Question", back_populates="prerequisite")

class Question(Base):
    __tablename__ = "questions"
    
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    prerequisite_id = Column(String, ForeignKey("prerequisites.id"))
    question_text = Column(Text, nullable=False)
    options = Column(JSON)  # List of options
    correct_answer = Column(Integer, nullable=False)
    explanation = Column(Text)
    difficulty = Column(String)  # beginner, intermediate, advanced
    topic = Column(String)
    is_ai_generated = Column(Boolean, default=True)
    created_at = Column(DateTime, default=datetime.utcnow)
    
    # Relationships
    prerequisite = relationship("Prerequisite", back_populates="questions")
    answers = relationship("UserAnswer", back_populates="question")

class Assessment(Base):
    __tablename__ = "assessments"
    
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    user_id = Column(UUID(as_uuid=True), ForeignKey("users.id"))
    goal = Column(String, nullable=False)
    target_level = Column(String)
    prerequisite_ids = Column(JSON)  # List of selected prerequisite IDs
    status = Column(String, default="active")  # active, completed, abandoned
    overall_score = Column(Float)
    total_questions = Column(Integer)
    correct_answers = Column(Integer)
    time_taken_minutes = Column(Integer)
    created_at = Column(DateTime, default=datetime.utcnow)
    completed_at = Column(DateTime)
    
    # Relationships
    user = relationship("User", back_populates="assessments")
    answers = relationship("UserAnswer", back_populates="assessment")
    results = relationship("AssessmentResult", back_populates="assessment")

class UserAnswer(Base):
    __tablename__ = "user_answers"
    
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    assessment_id = Column(UUID(as_uuid=True), ForeignKey("assessments.id"))
    question_id = Column(UUID(as_uuid=True), ForeignKey("questions.id"))
    selected_option = Column(Integer, nullable=False)
    is_correct = Column(Boolean)
    time_taken_seconds = Column(Integer)
    created_at = Column(DateTime, default=datetime.utcnow)
    
    # Relationships
    assessment = relationship("Assessment", back_populates="answers")
    question = relationship("Question", back_populates="answers")

class AssessmentResult(Base):
    __tablename__ = "assessment_results"
    
    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    assessment_id = Column(UUID(as_uuid=True), ForeignKey("assessments.id"))

    total_questions = Column(Integer)

    created_at = Column(DateTime, default=datetime.utcnow)
    
    # Relationships
    assessment = relationship("Assessment", back_populates="results")

class ProgressRecord(Base):
    __tablename__ = "progress_records"

    created_at = Column(DateTime, default=datetime.utcnow)

    user = relationship("User", back_populates="progress_records")

import openai
import json
import random
from typing import List, Dict, Any
from config import settings

# Configure OpenAI
openai.api_key = settings.openai_api_key

class AIService:
    """Service for AI-powered content generation"""
    
    @staticmethod
    async def generate_custom_levels(goal: str) -> List[Dict[str, Any]]:
        """Generate career levels using OpenAI"""
        
        prompt = f"""
        Generate 5 career levels for the role "{goal}" in JSON format.
        Each level should include:
        - id: snake_case unique identifier
        - name: Professional level name
        - description: 1-2 sentence description of responsibilities and expectations
        - experience_years: Expected years of experience (e.g., "0-2 years")
        - required_skills: Array of key skills needed (use snake_case ids)
        
        Progress from entry-level to senior/leadership positions.
        Focus on realistic career progression in the tech industry.
        
        Return only valid JSON array.
        """
        
        try:
            response = await openai.ChatCompletion.acreate(
                model=settings.openai_model,
                messages=[
                    {"role": "system", "content": "You are an expert career advisor and technical recruiter."},
                    {"role": "user", "content": prompt}
                ],
                temperature=0.3,
                max_tokens=1500
            )
            
            content = response.choices[0].message.content
            levels = json.loads(content)
            return levels
            
        except Exception as e:
            print(f"Error generating levels with AI: {e}")
            # Return fallback levels
            return AIService._get_fallback_levels(goal)
    
    @staticmethod
    async def generate_prerequisites(goal: str, target_level: str) -> List[Dict[str, Any]]:
        """Generate prerequisites using OpenAI"""
        
        prompt = f"""
        Generate prerequisites/skills needed for a {goal} at {target_level} level.
        
        Return 6-8 skills in JSON format with:
        - id: snake_case identifier
        - name: Human-readable skill name
        - description: What this skill involves (1-2 sentences)
        - importance: "Critical", "Important", or "Advanced"
        - related_topics: Array of specific subtopics/technologies
        
        Cover a mix of technical skills, tools, and soft skills relevant to the role.
        Adjust complexity based on the target level.
        
        Return only valid JSON array.
        """
        
        try:
            response = await openai.ChatCompletion.acreate(
                model=settings.openai_model,
                messages=[
                    {"role": "system", "content": "You are an expert in technical skills assessment and curriculum design."},
                    {"role": "user", "content": prompt}
                ],
                temperature=0.4,
                max_tokens=2000
            )
            
            content = response.choices[0].message.content
            prerequisites = json.loads(content)
            return prerequisites
            
        except Exception as e:
            print(f"Error generating prerequisites with AI: {e}")
            return AIService._get_fallback_prerequisites(goal)
    
    @staticmethod
    async def generate_questions(
        skill: str, 
        difficulty: str, 
        count: int = 5,
        topics: List[str] = None
    ) -> List[Dict[str, Any]]:
        """Generate assessment questions using OpenAI"""
        
        topics_context = f"Focus on these topics: {', '.join(topics)}" if topics else ""
        
        prompt = f"""
        Generate {count} multiple-choice questions for assessing {skill} at {difficulty} level.
        {topics_context}
        
        Each question should have:
        - question: Clear, specific question text
        - options: Array of 4 plausible options
        - correct: Index (0-3) of correct answer
        - explanation: Why the correct answer is right (2-3 sentences)
        - topic: Specific subtopic being tested
        
        Make questions practical and relevant to real-world scenarios.
        Difficulty levels:
        - beginner: Basic concepts, definitions, simple applications
        - intermediate: Applied knowledge, problem-solving, best practices
        - advanced: Complex scenarios, optimization, advanced concepts
        
        Return only valid JSON array.
        """
        
        try:
            response = await openai.ChatCompletion.acreate(
                model=settings.openai_model,
                messages=[
                    {"role": "system", "content": f"You are an expert instructor in {skill} creating assessment questions."},
                    {"role": "user", "content": prompt}
                ],
                temperature=0.5,
                max_tokens=3000
            )
            
            content = response.choices[0].message.content
            questions = json.loads(content)
            return questions
            
        except Exception as e:
            print(f"Error generating questions with AI: {e}")
            return AIService._get_fallback_questions(skill, difficulty, count)
    
    @staticmethod
    async def generate_recommendations(
        skill: str,
        current_level: str,
        score: float,
        weak_areas: List[str] = None
    ) -> List[str]:
        """Generate personalized learning recommendations"""
        
        weak_areas_context = f"Focus on these weak areas: {', '.join(weak_areas)}" if weak_areas else ""
        
        prompt = f"""
        Generate 4-5 personalized learning recommendations for someone learning {skill}.
        
        Context:
        - Current level: {current_level}
        - Recent assessment score: {score}%
        {weak_areas_context}
        
        Provide actionable, specific recommendations that include:
        - Specific resources or learning paths
        - Practical exercises or projects
        - Next steps for improvement
        
        Make recommendations encouraging and achievable.
        Return as a simple JSON array of strings.
        """
        
        try:
            response = await openai.ChatCompletion.acreate(
                model=settings.openai_model,
                messages=[
                    {"role": "system", "content": "You are a helpful learning advisor and mentor."},
                    {"role": "user", "content": prompt}
                ],
                temperature=0.6,
                max_tokens=1000
            )
            
            content = response.choices[0].message.content
            recommendations = json.loads(content)
            return recommendations
            
        except Exception as e:
            print(f"Error generating recommendations with AI: {e}")
            return AIService._get_fallback_recommendations(skill, current_level)
    
    # Fallback methods with predefined content
    @staticmethod
    def _get_fallback_levels(goal: str) -> List[Dict[str, Any]]:
        """Fallback levels when AI generation fails"""
        # Return the mock levels from the main file
        return [
            {
                "id": "junior",
                "name": f"Junior {goal}",
                "description": f"Entry-level {goal} with basic skills and supervised work",
                "experience_years": "0-2 years",
                "required_skills": ["fundamentals", "basic_tools"]
            },
            {
                "id": "senior",
                "name": f"Senior {goal}",
                "description": f"Experienced {goal} leading projects and mentoring others",
                "experience_years": "5+ years", 
                "required_skills": ["advanced_skills", "leadership", "architecture"]
            }
        ]
    
    @staticmethod
    def _get_fallback_prerequisites(goal: str) -> List[Dict[str, Any]]:
        """Fallback prerequisites when AI generation fails"""
        return [
            {
                "id": "fundamentals",
                "name": f"{goal} Fundamentals",
                "description": f"Core concepts and principles of {goal}",
                "importance": "Critical",
                "related_topics": ["basics", "principles", "concepts"]
            }
        ]
    
    @staticmethod
    def _get_fallback_questions(skill: str, difficulty: str, count: int) -> List[Dict[str, Any]]:
        """Fallback questions when AI generation fails"""
        questions = []
        for i in range(count):
            questions.append({
                "question": f"What is an important concept in {skill}?",
                "options": ["Option A", "Option B", "Option C", "Option D"],
                "correct": 0,
                "explanation": f"This tests understanding of {skill} concepts.",
                "topic": f"{skill}_basics"
            })
        return questions
    
    @staticmethod
    def _get_fallback_recommendations(skill: str, level: str) -> List[str]:
        """Fallback recommendations when AI generation fails"""
        return [
            f"Practice {skill} fundamentals daily",
            f"Work on {skill} projects to build experience",
            f"Join {skill} communities and forums",
            f"Take structured courses in {skill}",
            f"Seek feedback from {skill} experts"

# tasks.py
from celery import current_app as celery_app
from datetime import datetime, timedelta
import asyncio
from ai_service import AIService
from database import SessionLocal, Question, Assessment
from sqlalchemy import and_

@celery_app.task
def cleanup_expired_sessions():
    """Clean up expired assessment sessions"""
    db = SessionLocal()
    try:
        # Delete sessions older than 24 hours that are not completed
        cutoff_time = datetime.utcnow() - timedelta(hours=24)
        
        expired_sessions = db.query(Assessment).filter(
            and_(
                Assessment.created_at < cutoff_time,
                Assessment.status == 'active'
            )
        )
        
        count = expired_sessions.count()
        expired_sessions.update({'status': 'expired'})
        db.commit()
        
        return f"Cleaned up {count} expired sessions"
    
    except Exception as e:
        db.rollback()
        return f"Error cleaning up sessions: {str(e)}"
    finally:
        db.close()

@celery_app.task
def update_question_bank():
    """Update question bank with new AI-generated questions"""
    db = SessionLocal()
    try:
        # Get skills that need more questions
        skills_needing_questions = [
            'python', 'machine_learning', 'statistics', 
            'deep_learning', 'data_visualization'
        ]
        
        total_generated = 0
        
        for skill in skills_needing_questions:
            # Check current question count
            current_count = db.query(Question).filter(
                Question.prerequisite_id == skill
            ).count()
            
            if current_count < 50:  # Target 50 questions per skill
                # Generate new questions
                loop = asyncio.new_event_loop()
                asyncio.set_event_loop(loop)
                
                try:
                    questions = loop.run_until_complete(
                        AIService.generate_questions(skill, 'intermediate', 10)
                    )
                    
                    for q_data in questions:
                        question = Question(
                            prerequisite_id=skill,
                            question_text=q_data['question'],
                            options=q_data['options'],
                            correct_answer=q_data['correct'],
                            explanation=q_data['explanation'],
                            difficulty='intermediate',
                            topic=q_data.get('topic', 'general'),
                            is_ai_generated=True
                        )
                        db.add(question)
                    
                    total_generated += len(questions)
                finally:
                    loop.close()
        
        db.commit()
        return f"Generated {total_generated} new questions"
    
    except Exception as e:
        db.rollback()
        return f"Error updating question bank: {str(e)}"
    finally:
        db.close()

from logging.config import fileConfig
from sqlalchemy import engine_from_config, pool
from alembic import context
import os
import sys

# Add the project directory to Python path
sys.path.insert(0, os.path.dirname(os.path.dirname(__file__)))

from config import settings
from database import Base

# Alembic Config object
config = context.config

# Set the database URL
config.set_main_option('sqlalchemy.url', settings.database_url)

# Configure logging
if config.config_file_name is not None:
    fileConfig(config.config_file_name)

# Target metadata for autogenerate support
target_metadata = Base.metadata

def run_migrations_offline() -> None:
    """Run migrations in 'offline' mode."""
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()

def run_migrations_online() -> None:
    """Run migrations in 'online' mode."""
    connectable = engine_from_config(
        config.get_section(config.config_ini_section),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection, target_metadata=target_metadata
        )

        with context.begin_transaction():
            context.run_migrations()

if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

Workflow Implementation:
High-level workflow (one-line steps)

User enters Goal → click Generate Assessment.

System creates a mock assessment (uses goal → maps to required skills & topics).

Show required skills; user selects skills they’re confident in; remaining skills → stack.

User clicks Create Mock Test (skills) → system generates test from selected skills.

After/during test, show current level and available target levels (must be higher than current).

User picks a target level, clicks Continue → show topics (headings & subheadings) for that level.

User selects topics; clicks Create Mock Test (topics) → generate topic-focused mock test.

User completes test → show results, per-skill breakdown, level update, concrete recommendations.

Use the stack of unselected skills to propose future micro-tests and spaced-repetition sessions.
